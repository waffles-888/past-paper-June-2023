small_price = 5medium_price = 20large_price = 30signature_price = 2tracking_price = 5while True:    name = input("Please enter the customer's name: ")    if name == "":        print("The name cannot be left blank")        continue    breakwhile True:    address = input("Please enter the customer's address: ")    if address == "":        print("The address cannot be left blank")        continue    breakwhile True:    phone = input("Please enter the customer's phone number: ")    if not phone.isdigit() or len(phone) != 11:        print("The phone number must be 11 numbers long")        continue    breakwhile True:    number_of_parcels = input("How many parcels are being collected (1 to 6): ")    if not number_of_parcels.isdigit():        print("Please enter a whole number")        continue    number_of_parcels = int(number_of_parcels)    if number_of_parcels < 1 or number_of_parcels > 6:        print("The number of parcels must be between 1 and 6")        continue    breaktotal_cost = 0parcels_collected = 0receipt = []for parcel in range(1, number_of_parcels + 1):    print("Parcel", parcel)    while True:        try:            height = int(input("Enter the height in cm: "))            length = int(input("Enter the length in cm: "))            width = int(input("Enter the width in cm: "))            weight = float(input("Enter the weight in kg: "))        except ValueError:            print("Please enter numbers only, the height, length and width must be whole numbers")            continue        if height <= 0 or length <= 0 or width <= 0 or weight <= 0:            print("The measurements and weight must be more than 0")            continue        break    size = height + length + width    if size > 450 or weight > 30:        print("This parcel is too big or too heavy and cannot be collected")        receipt.append("Parcel " + str(parcel) + ": not collected (too big or too heavy) - £0.00")        continue    if size < 95:        size_price = small_price    elif size <= 150:        size_price = medium_price    else:        size_price = large_price    if weight <= 2:        weight_price = small_price    elif weight <= 15:        weight_price = medium_price    else:        weight_price = large_price    if size_price > weight_price:        parcel_price = size_price    else:        parcel_price = weight_price    parcel_cost = parcel_price    receipt.append("Parcel " + str(parcel) + ": size " + str(size) + "cm, weight " + str(weight) + "kg - £" + format(parcel_price, ".2f"))    while True:        signature = input("Do you want signature on delivery for £2? (Y/N): ").upper()        if signature == "Y" or signature == "N":            break        print("Please enter Y or N")    if signature == "Y":        parcel_cost = parcel_cost + signature_price        receipt.append("   Signature on delivery - £" + format(signature_price, ".2f"))    while True:        tracking = input("Do you want parcel tracking for £5? (Y/N): ").upper()        if tracking == "Y" or tracking == "N":            break        print("Please enter Y or N")    if tracking == "Y":        parcel_cost = parcel_cost + tracking_price        receipt.append("   Parcel tracking - £" + format(tracking_price, ".2f"))    receipt.append("   Parcel " + str(parcel) + " total - £" + format(parcel_cost, ".2f"))    total_cost = total_cost + parcel_cost    parcels_collected = parcels_collected + 1print("Name:", name)print("Address:", address)print("Phone number:", phone)print("Number of parcels to collect:", parcels_collected)for line in receipt:    print(line)print("----------------------------------------")print("Total cost of collection: £" + format(total_cost, ".2f"))
